cmake_minimum_required(VERSION 3.22)
project(gesture_boundary LANGUAGES CXX)

include("${GESTURE_FLAGS_FILE}")
add_compile_options(${GESTURE_COMPILE_FLAGS})
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(GESTURE_FORK)
    add_library(gesture_runtime INTERFACE)
    add_library(Valdi::HarmonyRuntime ALIAS gesture_runtime)
    set(gesture_backend_target valdi_harmony_arkui)
else()
    add_library(valdi_runtime_ohos INTERFACE)
    set(gesture_backend_target valdi_arkui_backend)
endif()
add_subdirectory("${GESTURE_BACKEND}" backend)

if(GESTURE_HOST_FIXTURE)
    add_executable(gesture_native "${GESTURE_HOST_FIXTURE}")
    target_compile_features(gesture_native PRIVATE cxx_std_20)
    target_include_directories(gesture_native PRIVATE "${GESTURE_BACKEND}")
    target_link_options(gesture_native PRIVATE -fsanitize=address,undefined -Wl,-dead_strip -Wl,-undefined,error)
    # Only the real gesture.cpp object is linked, independently of the manager.
    target_link_libraries(gesture_native PRIVATE "${GESTURE_SUPPORT_ARCHIVE}")
    if(NOT GESTURE_MODULE_SOURCE)
        set(GESTURE_MODULE_SOURCE "${GESTURE_BACKEND}/gesture.cpp")
    endif()
    target_sources(gesture_native PRIVATE "${GESTURE_MODULE_SOURCE}")
    if(EXISTS "${GESTURE_BACKEND}/gesture_coordinator.cpp")
        if(NOT GESTURE_COORDINATOR_SOURCE)
            set(GESTURE_COORDINATOR_SOURCE "${GESTURE_BACKEND}/gesture_coordinator.cpp")
        endif()
        target_sources(gesture_native PRIVATE "${GESTURE_COORDINATOR_SOURCE}")
    endif()
    if(EXISTS "${GESTURE_BACKEND}/gesture_admission.cpp")
        if(NOT GESTURE_ADMISSION_SOURCE)
            set(GESTURE_ADMISSION_SOURCE "${GESTURE_BACKEND}/gesture_admission.cpp")
        endif()
        target_sources(gesture_native PRIVATE "${GESTURE_ADMISSION_SOURCE}")
    endif()
endif()
