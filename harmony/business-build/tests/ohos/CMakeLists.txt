cmake_minimum_required(VERSION 3.22)
project(valdi_clientsql_cross_proof LANGUAGES C CXX)

# Import already-built real OHOS static archives for an isolated ClientSQL
# build. Production consumers add business-build after runtime-build.
# Recover the real compile interface, including overlays and generated config,
# from the Runtime.cpp compilation instead of inventing runtime declarations.
set(VALDI_CLIENTSQL_RUNTIME_BUILD_DIR "" CACHE PATH "Previously built runtime directory for this ABI")
if(NOT EXISTS "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}/libvaldi_runtime_ohos_link_check.so")
  message(FATAL_ERROR "Build valdi_runtime_ohos_link_check for the selected ABI first.")
endif()
file(STRINGS "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}/CMakeCache.txt" _arch_line REGEX "^OHOS_ARCH:[^=]+=")
if(NOT _arch_line MATCHES "=${OHOS_ARCH}$")
  message(FATAL_ERROR "The imported runtime ABI differs from OHOS_ARCH.")
endif()
file(READ "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}/compile_commands.json" _commands)
string(JSON _count LENGTH "${_commands}")
math(EXPR _last "${_count} - 1")
foreach(_index RANGE ${_last})
  string(JSON _file GET "${_commands}" ${_index} file)
  if(_file MATCHES "/valdi/runtime/Runtime.cpp$")
    string(JSON _command GET "${_commands}" ${_index} command)
    break()
  endif()
endforeach()
if(NOT _command)
  message(FATAL_ERROR "No actual Runtime.cpp compile interface in the runtime build.")
endif()
separate_arguments(_arguments UNIX_COMMAND "${_command}")
set(_interface "")
set(_system_includes "")
set(_system_include_next OFF)
foreach(_argument IN LISTS _arguments)
  if(_system_include_next)
    list(APPEND _system_includes "${_argument}")
    set(_system_include_next OFF)
  elseif(_argument STREQUAL "-isystem")
    set(_system_include_next ON)
  elseif(_argument MATCHES "^(-I|-D)")
    list(APPEND _interface "${_argument}")
  endif()
endforeach()
# Recover the existing Ninja link closure so every dependency remains the same
# ABI and toolchain as the runtime. This is test-driver-only: no guessed library
# list, Apple binary, or unresolved-symbol bypass enters the production wrapper.
file(STRINGS "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}/build.ninja" _link_line
  REGEX "^  LINK_LIBRARIES = .*--whole-archive.*libvaldi_runtime_ohos.a")
list(LENGTH _link_line _link_count)
if(NOT _link_count EQUAL 1)
  message(FATAL_ERROR "Expected exactly one strict runtime link closure in build.ninja.")
endif()
string(REGEX REPLACE "^  LINK_LIBRARIES = " "" _link_command "${_link_line}")
separate_arguments(_link_arguments UNIX_COMMAND "${_link_command}")
set(_dependencies "")
foreach(_argument IN LISTS _link_arguments)
  if(_argument STREQUAL "libvaldi_runtime_ohos.a" OR _argument MATCHES "^-Wl,")
    continue()
  elseif(_argument MATCHES "\\.a$")
    get_filename_component(_archive "${_argument}" ABSOLUTE BASE_DIR "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}")
    if(NOT EXISTS "${_archive}")
      message(FATAL_ERROR "Runtime dependency is not built: ${_archive}")
    endif()
    list(APPEND _dependencies "${_archive}")
  elseif(_argument MATCHES "^-l")
    list(APPEND _dependencies "${_argument}")
  else()
    message(FATAL_ERROR "Unrecognized runtime link argument: ${_argument}")
  endif()
endforeach()
add_library(valdi_runtime_ohos STATIC IMPORTED GLOBAL)
set_target_properties(valdi_runtime_ohos PROPERTIES
  IMPORTED_LOCATION "${VALDI_CLIENTSQL_RUNTIME_BUILD_DIR}/libvaldi_runtime_ohos.a"
  INTERFACE_COMPILE_OPTIONS "${_interface}"
  INTERFACE_LINK_LIBRARIES "${_dependencies}")
target_include_directories(valdi_runtime_ohos SYSTEM INTERFACE ${_system_includes})
set(VALDI_CLIENTSQL_LINK_CHECK ON CACHE BOOL "" FORCE)
add_subdirectory(../.. business)
